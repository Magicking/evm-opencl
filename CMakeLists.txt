cmake_minimum_required(VERSION 3.24)
project(fastevm LANGUAGES C CXX)

option(FASTEVM_GPU "Build the CUDA backend" ON)
option(FASTEVM_TESTS "Build unit tests" ON)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")

set(FASTEVM_SOURCES src/runtime/engine.cpp src/runtime/state_cache.cpp src/capi/flat.cpp)

if(FASTEVM_GPU)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
      set(CMAKE_CUDA_ARCHITECTURES 89)
    endif()
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 20)
    list(APPEND FASTEVM_SOURCES src/runtime/gpu_backend.cu)
  else()
    message(WARNING "CUDA compiler not found: building CPU backend only")
    set(FASTEVM_GPU OFF)
  endif()
endif()

add_library(fastevm SHARED ${FASTEVM_SOURCES})
add_library(fastevm_static STATIC ${FASTEVM_SOURCES})
foreach(t fastevm fastevm_static)
  target_include_directories(${t} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_options(${t} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall -Wno-unused-function -Wno-attributes -Wno-stringop-overflow>)
  if(FASTEVM_GPU)
    target_compile_definitions(${t} PRIVATE FASTEVM_WITH_CUDA=1)
    target_compile_options(${t} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr -lineinfo -Xptxas=-v>)
    set_target_properties(${t} PROPERTIES CUDA_RESOLVE_DEVICE_SYMBOLS ON)
    find_package(CUDAToolkit REQUIRED)
    target_link_libraries(${t} PUBLIC CUDA::cudart_static)
  endif()
  target_link_libraries(${t} PUBLIC pthread)
endforeach()
set_target_properties(fastevm_static PROPERTIES OUTPUT_NAME fastevm)

add_executable(fastevm-run tools/fastevm-run/main.cpp)
target_link_libraries(fastevm-run PRIVATE fastevm_static)

add_executable(fastevm-bench tools/fastevm-bench/main.cpp)
target_link_libraries(fastevm-bench PRIVATE fastevm_static)

if(FASTEVM_TESTS)
  find_package(GTest)
  if(GTest_FOUND)
    enable_testing()
    add_executable(fastevm-tests tests/test_primitives.cpp tests/test_interpreter.cpp tests/test_state_cache.cpp)
    target_link_libraries(fastevm-tests PRIVATE fastevm_static GTest::gtest_main)
    target_include_directories(fastevm-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
    include(GoogleTest)
    gtest_discover_tests(fastevm-tests)
  endif()
endif()

install(TARGETS fastevm fastevm_static LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES include/fastevm.h DESTINATION include)
